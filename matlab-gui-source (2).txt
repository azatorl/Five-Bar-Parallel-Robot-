        % Code that executes after component creation
        function startupFcn(app)
          app.UIFigure.UserData = struct();

    % ============================================================
    % Store all project settings in app.UIFigure.UserData
    % ============================================================

    app.UIFigure.UserData.port = "COM7";
    app.UIFigure.UserData.baud = 9600;

    app.UIFigure.UserData.modeLeft = 1;
    app.UIFigure.UserData.modeRight = 2;

    app.UIFigure.UserData.Px = 50;
    app.UIFigure.UserData.PyHome = 80;
    app.UIFigure.UserData.PyMin = 80;
    app.UIFigure.UserData.PyMax = 120;

    app.UIFigure.UserData.isConnected = false;
    app.UIFigure.UserData.serialObj = [];

    % ============================================================
    % Initial UI values
    % ============================================================

    app.PyTargetmmEditField.Value = app.UIFigure.UserData.PyHome;

   
    app.theta1EditField.Value = 0;
    app.theta2EditField.Value = 0;
    app.theta3EditField.Value = 0;
    app.theta4EditField.Value = 0;

    title(app.UIAxes, 'Five-Bar Robot');
    xlabel(app.UIAxes, 'Px (mm)');
    ylabel(app.UIAxes, 'Py (mm)');
    grid(app.UIAxes, 'on');
    axis(app.UIAxes, 'equal');

    % ============================================================
    % Try Arduino connection automatically
    % ============================================================

    try
        app.UIFigure.UserData.serialObj = serialport(app.UIFigure.UserData.port, app.UIFigure.UserData.baud);
        configureTerminator(app.UIFigure.UserData.serialObj, "LF");
        pause(2);

        app.UIFigure.UserData.isConnected = true;

        writeline(app.UIFigure.UserData.serialObj, "HOME");

    catch
        app.UIFigure.UserData.isConnected = false;
    end

    % ============================================================
    % Draw home position
    % ============================================================

    targetPy = app.UIFigure.UserData.PyHome;

    result = fivebarIK_project( ...
        app.UIFigure.UserData.Px, ...
        targetPy, ...
        app.UIFigure.UserData.modeLeft, ...
        app.UIFigure.UserData.modeRight);

    if result.ok
        drawFivebarOnAxes(app.UIAxes, app.UIFigure.UserData.Px, targetPy, result.A, result.B);

        title(app.UIAxes, sprintf('Five-Bar Robot | Px = %.1f mm | Py = %.1f mm', ...
            app.UIFigure.UserData.Px, targetPy));

        app.theta1EditField.Value = result.theta1;
        app.theta2EditField.Value = result.theta2;
        app.theta3EditField.Value = result.theta3;
        app.theta4EditField.Value = result.theta4;
   
end 
        end

        % Button pushed function: MoveButton
        function MoveButtonPushed(app, event)
        

    % ============================================================
    % Read target Py
    % ============================================================

    targetPy = round(app.PyTargetmmEditField.Value);

    % ============================================================
    % Workspace limit control
    % ============================================================

    if targetPy < app.UIFigure.UserData.PyMin
        targetPy = app.UIFigure.UserData.PyMin;

        uialert(app.UIFigure, ...
            'Minimum Py is 80 mm. Target was limited to 80 mm.', ...
            'Workspace Limit');
    end

    if targetPy > app.UIFigure.UserData.PyMax
        targetPy = app.UIFigure.UserData.PyMax;

        uialert(app.UIFigure, ...
            'Maximum Py is 120 mm. Target was limited to 120 mm.', ...
            'Workspace Limit');
    end

    app.PyTargetmmEditField.Value = targetPy;

    % ============================================================
    % Inverse kinematics and drawing
    % ============================================================

    result = fivebarIK_project( ...
        app.UIFigure.UserData.Px, ...
        targetPy, ...
        app.UIFigure.UserData.modeLeft, ...
        app.UIFigure.UserData.modeRight);

    if ~result.ok
        uialert(app.UIFigure, result.message, 'IK Error');
        return;
    end

    drawFivebarOnAxes(app.UIAxes, app.UIFigure.UserData.Px, targetPy, result.A, result.B);

    title(app.UIAxes, sprintf('Five-Bar Robot | Px = %.1f mm | Py = %.1f mm', ...
        app.UIFigure.UserData.Px, targetPy));

    app.theta1EditField.Value = result.theta1;
    app.theta2EditField.Value = result.theta2;
    app.theta3EditField.Value = result.theta3;
    app.theta4EditField.Value = result.theta4;

    % ============================================================
    % Send command to Arduino
    % ============================================================

    if app.UIFigure.UserData.isConnected
        try
            packet = "Y:" + string(targetPy);
            writeline(app.UIFigure.UserData.serialObj, packet);

        catch
            app.UIFigure.UserData.isConnected = false;

            uialert(app.UIFigure, ...
                'Arduino connection error. The robot was drawn only.', ...
                'Serial Error');
        end
    else
        uialert(app.UIFigure, ...
            'Arduino is not connected. The robot was drawn only.', ...
            'Serial Warning');
    

end
        end

        % Button pushed function: HomeButton
        function HomeButtonPushed(app, event)
      
    targetPy = 80;
    app.PyTargetmmEditField.Value = targetPy;

    result = fivebarIK_project( ...
        app.UIFigure.UserData.Px, ...
        targetPy, ...
        app.UIFigure.UserData.modeLeft, ...
        app.UIFigure.UserData.modeRight);

    if ~result.ok
        uialert(app.UIFigure, result.message, 'IK Error');
        return;
    end

    drawFivebarOnAxes(app.UIAxes, app.UIFigure.UserData.Px, targetPy, result.A, result.B);

    title(app.UIAxes, sprintf('Home Position | Px = %.1f mm | Py = %.1f mm', ...
        app.UIFigure.UserData.Px, targetPy));

    app.theta1EditField.Value = result.theta1;
    app.theta2EditField.Value = result.theta2;
    app.theta3EditField.Value = result.theta3;
    app.theta4EditField.Value = result.theta4;

    try
        if isempty(app.UIFigure.UserData.serialObj) || ~isvalid(app.UIFigure.UserData.serialObj)
            app.UIFigure.UserData.serialObj = serialport(app.UIFigure.UserData.port, app.UIFigure.UserData.baud);
            configureTerminator(app.UIFigure.UserData.serialObj, "LF");
            pause(3);
            app.UIFigure.UserData.isConnected = true;
        end

        writeline(app.UIFigure.UserData.serialObj, "HOME");

    catch ME
        app.UIFigure.UserData.isConnected = false;
        app.UIFigure.UserData.serialObj = [];

        uialert(app.UIFigure, ...
            "Arduino connection error: " + string(ME.message), ...
            'Serial Error');
    
end
        end
    end
